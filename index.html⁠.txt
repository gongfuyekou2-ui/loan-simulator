    let myChart = null;

    function calculate() {
      const amountMan = parseFloat(document.getElementById('borrowAmount').value) || 0;
      const annualRate = parseFloat(document.getElementById('interestRate').value) || 0;
      const years = parseInt(document.getElementById('loanYears').value) || 0;
      const type = document.getElementById('repaymentType').value;

      const P = amountMan * 10000; // 円単位
      const r = (annualRate / 100) / 12; // 月利
      const n = years * 12; // 総返済回数

      if (P <= 0 || n <= 0) return;

      let monthly = 0;
      let totalPay = 0;
      let totalInt = 0;
      let schedule = [];

      if (type === 'principal-interest') {
        // 元利均等返済
        if (r === 0) {
          monthly = P / n;
        } else {
          monthly = P * (r * Math.pow(1 + r, n)) / (Math.pow(1 + r, n) - 1);
        }
        totalPay = monthly * n;
        totalInt = totalPay - P;

        // 年別推移の計算
        let balance = P;
        for (let y = 1; y <= years; y++) {
          let yearInterest = 0;
          let yearPrincipal = 0;
          for (let m = 1; m <= 12; m++) {
            let intPayment = balance * r;
            let prinPayment = monthly - intPayment;
            yearInterest += intPayment;
            yearPrincipal += prinPayment;
            balance -= prinPayment;
          }
          if (balance < 0) balance = 0;
          schedule.push({ year: y, yearPay: monthly * 12, yearInt: yearInterest, balance: balance });
        }
      } else {
        // 元金均等返済
        const principalMonthly = P / n;
        let balance = P;
        let sumPay = 0;

        for (let y = 1; y <= years; y++) {
          let yearInterest = 0;
          for (let m = 1; m <= 12; m++) {
            let intPayment = balance * r;
            yearInterest += intPayment;
            balance -= principalMonthly;
          }
          let yearPay = (principalMonthly * 12) + yearInterest;
          sumPay += yearPay;
          if (y === 1) monthly = (principalMonthly + (P * r)); // 初月返済額を表示用に使用
          if (balance < 0) balance = 0;
          schedule.push({ year: y, yearPay: yearPay, yearInt: yearInterest, balance: balance });
        }
        totalPay = sumPay;
        totalInt = totalPay - P;
      }

      // UI反映
      document.getElementById('monthlyPayment').innerText = Math.round(monthly).toLocaleString() + ' 円';
      document.getElementById('totalPayment').innerText = Math.round(totalPay / 10000).toLocaleString() + ' 万円';
      document.getElementById('totalInterest').innerText = Math.round(totalInt / 10000).toLocaleString() + ' 万円';

      updateChart(P, totalInt);
      updateTable(schedule);
    }

    function updateChart(principal, interest) {
      const ctx = document.getElementById('pieChart').getContext('2d');
      if (myChart) myChart.destroy();

      myChart = new Chart(ctx, {
        type: 'doughnut',
        data: {
          labels: ['元金', '利息'],
          datasets: [{
            data: [principal, interest],
            backgroundColor: ['#38bdf8', '#ef4444'],
            borderWidth: 0
          }]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          plugins: {
            legend: {
              position: 'bottom',
              labels: { color: '#f8fafc' }
            }
          }
        }
      });
    }

    function updateTable(schedule) {
      const tbody = document.querySelector('#scheduleTable tbody');
      tbody.innerHTML = '';

      schedule.forEach(row => {
        const tr = document.createElement('tr');
        tr.innerHTML = `
          <td>${row.year}年目</td>
          <td>${Math.round(row.yearPay / 10000).toLocaleString()}万円</td>
          <td>${Math.round(row.yearInt / 10000).toLocaleString()}万円</td>
          <td>${Math.round(row.balance / 10000).toLocaleString()}万円</td>
        `;
        tbody.appendChild(tr);
      });
    }

    function downloadCSV() {
      const amount = document.getElementById('borrowAmount').value;
      const rate = document.getElementById('interestRate').value;
      const years = document.getElementById('loanYears').value;
      let csvContent = "data:text/csv;charset=utf-8,";
      csvContent += `条件: 借入額${amount}万円, 金利${rate}%, 期間${years}年\n`;
      csvContent += "経過年,年間返済額(円),内利息(円),残高(円)\n";

      const rows = document.querySelectorAll('#scheduleTable tbody tr');
      rows.forEach(r => {
        const cols = r.querySelectorAll('td');
        const rowData = Array.from(cols).map(c => c.innerText.replace(/万円|年目|,/g, '')).join(',');
        csvContent += rowData + "\n";
      });

      const encodedUri = encodeURI(csvContent);
      const link = document.createElement("a");
      link.setAttribute("href", encodedUri);
      link.setAttribute("download", "住宅ローン返済シミュレーション.csv");
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
    }

    // 初期計算を実行
    calculate();
  </script>
